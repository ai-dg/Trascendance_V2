# General:

    Folders:

# Frontend:

    Commands:
        - pnpm run build:css // run the tailwind build
        - pnpm run dev:css // run tailwind build in watch mode for dev

    Folders:
        - locales   // for dictionary in language manager
        - config    // all the config files
        - public    // public files for frontend: JS / TS / Types / CSS
        - views     // ejs tout court
